% SLIDE 02: THE PROBLEM — VIBE CODING VS SDD & SOFTWARE 3.0 VERIFICATION LOOP
% ============================================================================
\slideshell{02 / 12}{MOTIVATION \& SOFTWARE 3.0 VERIFICATION}{From ``Vibe Coding'' to Agentic Engineering: Validation \& Human Interaction}{Why generation is free, human verification is the bottleneck, and agents need a short leash + immutable verifiers}{
  \bcard{0}{0}{7.1}{3.9}{gbgred}{gred}{
    {\fontsize{16}{20}\selectfont\bfseries\textcolor{gred}{Anti-Pattern: The Vibe Coding Amnesia Loop}}\\[8pt]
    \textbf{1. Silent Assumptions:} ``Overeager junior'' agent guesses ambiguous edge cases without asking the human.\\[5pt]
    \textbf{2. 1,000-Line Bloat \& Collateral Edits:} Agent over-engineers abstractions and mutates adjacent working code (e.g.\ FIFO $\to$ LRU).\\[5pt]
    \textbf{3. Sycophantic Self-Testing:} When the same unconstrained prompt writes code and tests, it weakens assertions so broken code passes.\\[5pt]
    \textbf{4. Unreviewable PR Slop:} 2,000-line diffs overwhelm human reviewers.
  }
  \bcard{7.6}{0}{7.1}{3.9}{gbggreen}{ggreen}{
    {\fontsize{16}{20}\selectfont\bfseries\textcolor{ggreen}{Solution: 4 Human Gates \& 4 Validation Layers}}\\[8pt]
    \textbf{1. Think Before Coding:} Sandboxed \code{/conductor:newTrack} Q\&A surfaces trade-offs before any code is written.\\[5pt]
    \textbf{2. Spec-First Steering (\code{program.md} Pattern):} Human approves 50-line RFC 2119 + Gherkin \code{spec.md} before \code{/conductor:implement}.\\[5pt]
    \textbf{3. Surgical Scope \& Short Leash:} Lock \code{Out of Scope} + \code{tech-stack.md}; verify at small Phase Checkpoints \& Stacked PR batches.\\[5pt]
    \textbf{4. Immutable Verifiers (\code{chmod -w}):} 1:1 \code{test\_reqXXXX\_*} TDD + read-only adversarial tests (Karpathy \code{autoresearch} pattern).
  }

  \bcard{0}{-4.15}{14.7}{2.15}{gbggray}{gblue}{
    {\fontsize{15}{19}\selectfont\bfseries\textcolor{gblue}{Karpathy's Software 3.0 Law: Speed Up the Generation--Verification Loop (``Iron Man Suit, Not Unsupervised Robot'')}}\\[6pt]
    $\bullet$ \textbf{4 Behavioral Agent Rules (in \code{conductor/workflow.md}):} \textbf{(1) Think Before Coding} (ask, don't guess) $\cdot$ \textbf{(2) Simplicity First} (minimal TDD code) $\cdot$ \textbf{(3) Surgical Changes} (honor \code{Out of Scope}) $\cdot$ \textbf{(4) Goal-Driven Execution} (loop against objective verifiers: \code{openspec validate {-\/-}strict}, \code{test\_reqXXXX\_*}, and \code{chmod -w} Clean-Room Rebuild Tests).
  }
}

% SLIDE 03: THE 3 LEVELS OF SDD MATURITY
% ============================================================================
\slideshell{03 / 12}{TAXONOMY \& MATURITY MODEL}{The 3 Levels of Spec-Driven Development Maturity}{Moving from disposable pre-coding specs to living delta synchronization and deterministic clean-room rebuilds}{
  \bcard{0}{0}{4.7}{4.3}{gbggray}{gborder}{
    {\fontsize{16}{20}\selectfont\bfseries\textcolor{gmuted}{Level 1: Spec-First}}\\[6pt]
    \textbf{``Write a spec before coding, then discard it.''}\\[8pt]
    $\bullet$ Engineers draft a design doc or prompt spec to generate v1 of a feature.\\[5pt]
    $\bullet$ Subsequent bugfixes and hotfixes edit source code directly.\\[5pt]
    $\bullet$ \textbf{Failure Mode:} Within 2 sprints, the spec is stale documentation that misleads future agents.
  }
  \bcard{5.0}{0}{4.7}{4.3}{gbgblue}{gblue}{
    {\fontsize{16}{20}\selectfont\bfseries\textcolor{gblue}{Level 2: Spec-Anchored}}\\[6pt]
    \textbf{``Living specs synchronized on every PR.''}\\[8pt]
    $\bullet$ Powered by \textbf{OpenSpec} (\code{openspec/specs/}) and \textbf{Conductor} (\code{conductor/tracks/}).\\[5pt]
    $\bullet$ Every change proposes explicit \code{ADDED / MODIFIED / REMOVED / RENAMED} requirement deltas.\\[5pt]
    $\bullet$ \textbf{Guarantee:} CI and \code{/opsx:verify} block PRs whenever code and specs diverge.
  }
  \bcard{10.0}{0}{4.7}{4.3}{gbggreen}{ggreen}{
    {\fontsize{16}{20}\selectfont\bfseries\textcolor{ggreen}{Level 3: Spec-as-Source (SSOT)}}\\[6pt]
    \textbf{``The spec can regenerate the system from scratch.''}\\[8pt]
    $\bullet$ Powered by the canonical \textbf{8-Section \code{SPEC.md}} blueprint.\\[5pt]
    $\bullet$ Zero raw customer secrets or hardcoded environment IDs.\\[5pt]
    $\bullet$ \textbf{Guarantee:} Passes the \textbf{Clean-Room Rebuild Test} in an empty \code{/tmp/} directory against locked adversarial tests.
  }

  \bcard{0}{-4.55}{14.7}{1.75}{gbgyellow}{gyellow}{
    {\fontsize{14}{18}\selectfont\bfseries\textcolor{gdark}{Universal Decision Rule: Which Level \& Tool Do I Pick Today?}}\\[4pt]
    $\bullet$ \textbf{Single-repo feature, bugfix, or TDD refactor?} Use \textbf{Conductor} (\code{/conductor:newTrack} $\to$ \code{/conductor:implement}). \quad
    $\bullet$ \textbf{Cross-cutting capability contract or brownfield delta?} Layer \textbf{OpenSpec} (\code{/opsx:propose} $\to$ \code{/opsx:archive}). \quad
    $\bullet$ \textbf{Re-architecting or extracting a portable microservice blueprint?} Author an \textbf{8-Section \code{SPEC.md}} and run the Rebuild Test.
  }
}

\slideshell{05 / 12}{ENGINE 2: OPENSPEC}{OpenSpec: Two-Plane Architecture \& Deterministic Delta Algebra}{Isolating current system truth (\code{openspec/specs/}) from active work-in-progress proposals (\code{openspec/changes/})}{
  \bcard{0}{0}{7.1}{4.0}{gbggreen}{ggreen}{
    {\fontsize{16}{20}\selectfont\bfseries\textcolor{ggreen}{Plane 1: \code{openspec/specs/<capability>/spec.md}}}\\[6pt]
    \textbf{The Living System of Record (Current Truth):}\\[4pt]
    $\bullet$ Organized by architectural capability (\code{rate-limiter}, \code{lease-manager}, \code{enrollment}), never by sprint ticket.\\[4pt]
    $\bullet$ Every requirement has a stable ID (\code{REQ-0001}) and RFC 2119 normative verb (\code{SHALL}, \code{MUST}, \code{MUST NOT}).\\[4pt]
    $\bullet$ Every requirement is paired with 4-hashtag \code{\#\#\#\# Scenario:} blocks using \code{GIVEN / WHEN / THEN}.
  }
  \bcard{7.6}{0}{7.1}{4.0}{gbgblue}{gblue}{
    {\fontsize{16}{20}\selectfont\bfseries\textcolor{gblue}{Plane 2: \code{openspec/changes/<change-id>/}}}\\[6pt]
    \textbf{Isolated Delta Bundle (\code{proposal.md}, \code{design.md}, \code{tasks.md}, \code{specs/}):}\\[4pt]
    $\bullet$ Expresses exact mutations via 4 delta headers: \code{ADDED}, \code{MODIFIED}, \code{REMOVED} (requires \textbf{Reason} \& \textbf{Migration}), and \code{RENAMED}.\\[4pt]
    $\bullet$ \textbf{Deterministic 4-Step Archive Merge (\code{/opsx:archive}):}\\[2pt]
    \code{1. Pre-Validate $\to$ 2. RENAMED $\to$ 3. REMOVED $\to$ 4. MODIFIED $\to$ 5. ADDED}.
  }

  \bcard{0}{-4.25}{14.7}{2.05}{gbgyellow}{gyellow}{
    {\fontsize{14}{18}\selectfont\bfseries\textcolor{gdark}{The 99\% / 1\% OpenSpec Rule \& The ``Core 4'' Daily Loop}}\\[4pt]
    $\bullet$ \textbf{1\% Terminal CLI:} Run \code{openspec init} once per repo (and \code{openspec validate --all --strict --json} in CI).\\[3pt]
    $\bullet$ \textbf{99\% AI Slash Commands:} Drive all daily work through \textbf{\code{/opsx:explore}} (read-only investigation) $\to$ \textbf{\code{/opsx:propose}} (generate \code{proposal.md}, \code{specs/}, \code{design.md}, \code{tasks.md}) $\to$ \textbf{\code{/opsx:apply}} (implement tasks) $\to$ \textbf{\code{/opsx:verify}} \& \textbf{\code{/opsx:archive}} (audit \& merge deltas).
  }
}

  }
